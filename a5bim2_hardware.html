from microbit import *

# Variáveis
A = False
B = False
operador = 0  # 0=AND, 1=OR, 2=XOR, 3=NAND, 4=NOR

def desenhar_display():
    display.clear()
    
    # Primeira linha: A e B
    if A:
        display.set_pixel(0, 0, 9)
        display.set_pixel(0, 1, 9)
    if B:
        display.set_pixel(4, 0, 9)
        display.set_pixel(4, 1, 9)
    
    # Mostra o operador atual
    op_text = ["AND", "OR ", "XOR", "NAND", "NOR "]
    display.scroll(op_text[operador])
    
    # Calcula e mostra resultado
    resultado = False
    if operador == 0:  # AND
        resultado = A and B
    elif operador == 1:  # OR
        resultado = A or B
    elif operador == 2:  # XOR
        resultado = (A or B) and not (A and B)
    elif operador == 3:  # NAND
        resultado = not (A and B)
    elif operador == 4:  # NOR
        resultado = not (A or B)
    
    # Mostra resultado
    if resultado:
        display.set_pixel(2, 2, 9)
        display.set_pixel(2, 3, 9)
        display.set_pixel(3, 2, 9)
        display.set_pixel(3, 3, 9)
    else:
        display.set_pixel(2, 2, 0)

# Loop principal
while True:
    if button_a.was_pressed():
        A = not A
        desenhar_display()
    
    if button_b.was_pressed():
        B = not B
        desenhar_display()

        operador = (operador + 1) % 5
        desenhar_display()

let pontuacao = 0
let tentativas = 0
let resposta = 0
let numero1 = 0
let numero2 = 0
let acertou = false

function sortearNumeros() {
    numero1 = Math.randomRange(2, 9)
    numero2 = Math.randomRange(2, 9)
}

function fazerPergunta() {
    sortearNumeros()
    let resultadoCorreto = numero1 * numero2
    tentativas = 0
    acertou = false
    
    // Mostra a pergunta na matriz
    basic.showString("" + numero1 + "x" + numero2 + "=?")
    
    while (tentativas < 3 && !acertou) {
        let respostaDigitada = obterResposta()
        if (respostaDigitada == resultadoCorreto) {
            acertou = true
            pontuacao += 10
            mostrarFeedback(true)
            basic.showString("ACERTOU!")
        } else {
            tentativas += 1
            mostrarFeedback(false)
            if (tentativas < 3) {
                basic.showString("TENTE NOV")
            }
        }
    }
    
    if (!acertou) {
        basic.showString("E " + resultadoCorreto)
        if (pontuacao >= 5) {
            pontuacao -= 5
        }
    }
}

function obterResposta(): number {
    let numero = 0
    let confirmado = false
    
    while (!confirmado) {
        if (input.buttonIsPressed(Button.A)) {
            numero = Math.min(numero + 1, 99)
            basic.showNumber(numero)
            basic.pause(200)
        }
        if (input.buttonIsPressed(Button.B)) {
            numero = Math.max(numero - 1, 0)
            basic.showNumber(numero)
            basic.pause(200)
        }
        if (input.buttonIsPressed(Button.AB)) {
            confirmado = true
        }
        basic.pause(50)
    }
    return numero
}

function mostrarFeedback(acertou: boolean) {
    if (acertou) {
        basic.showIcon(IconNames.Heart)
        for (let i = 0; i < 3; i++) {
            basic.showIcon(IconNames.Heart)
            basic.pause(200)
            basic.showIcon(IconNames.SmallHeart)
            basic.pause(200)
        }
    } else {
        basic.showIcon(IconNames.No)
        for (let i = 0; i < 3; i++) {
            basic.showIcon(IconNames.No)
            basic.pause(100)
            basic.clearScreen()
            basic.pause(100)
        }
    }
}

function mostrarPlacar() {
    basic.showString("P:" + pontuacao)
    let estrelas = Math.floor(pontuacao / 10)
    if (estrelas > 5) estrelas = 5
    display.clear()
    for (let i = 0; i < estrelas; i++) {
        display.set_pixel(i * 2, 2, 9)
        display.set_pixel(i * 2 + 1, 2, 9)
    }
    basic.pause(2000)
}

function lerLuz(): number {
    // O micro:bit usa os LEDs da matriz para medir luz
    // Retorna 0-255 (escuro-claro)
    return input.lightLevel()
}

function mostrarLuz() {
    let nivel = lerLuz()
    let barras = Math.map(nivel, 0, 255, 0, 5)
    display.clear()
    for (let i = 0; i < barras; i++) {
        for (let j = 0; j < 5; j++) {
            display.set_pixel(i, j, 9)
        }
    }
}

function modoVampiro() {
    let luz = lerLuz()
    if (luz < limiar) {
        // ESCURO - anda
        basic.showIcon(IconNames.ArrowE)
        basic.pause(200)
        // Pisca os olhos (LEDs no canto superior)
        for (let i = 0; i < 3; i++) {
            display.set_pixel(0, 0, 9)
            display.set_pixel(4, 0, 9)
            basic.pause(100)
            display.set_pixel(0, 0, 0)
            display.set_pixel(4, 0, 0)
            basic.pause(100)
        }
        basic.showIcon(IconNames.ArrowE)
    } else {
        // CLARO - não anda
        basic.showIcon(IconNames.Skull)
        for (let i = 0; i < 5; i++) {
            basic.showIcon(IconNames.Skull)
            basic.pause(150)
            display.clear()
            basic.pause(150)
        }
    }
}

function modoCalibracao() {
    let luz = lerLuz()
    basic.showNumber(luz)
    if (input.buttonIsPressed(Button.A)) {
        limiar = Math.min(limiar + 5, 255)
        basic.showString("T:" + limiar)
        basic.pause(200)
    }
    if (input.buttonIsPressed(Button.B)) {
        limiar = Math.max(limiar - 5, 0)
        basic.showString("T:" + limiar)
        basic.pause(200)
    }
}

let limiarSom = 50
let ultimoSom = 0

function medirSom(): number {
    // Retorna 0-255 (silêncio-grito)
    return input.soundLevel()
}

function detectarPalma(): boolean {
    let som = medirSom()
    // Detecta pico de som
    if (som > limiarSom && ultimoSom < limiarSom) {
        ultimoSom = som
        return true
    }
    ultimoSom = som
    return false
}

// Modos de reação
function reagirPalma() {
    // Efeito de onda na matriz
    for (let i = 0; i < 5; i++) {
        display.set_pixel(i, 2, 9)
        basic.pause(50)
        display.set_pixel(i, 1, 9)
        display.set_pixel(i, 3, 9)
        basic.pause(50)
    }
    basic.pause(200)
    display.clear()
}

function mostrarVolume() {
    let nivel = medirSom()
    let barras = Math.map(nivel, 0, 255, 0, 5)
    display.clear()
    for (let i = 0; i < barras; i++) {
        for (let j = 0; j < 5; j++) {
            display.set_pixel(i, j, 9)
        }
    }
}

// Programa principal
basic.showString("SOM")

while (true) {
    if (detectarPalma()) {
        reagirPalma()
        basic.showString("PALMA!")
    }
    // Mostra volume em tempo real
    mostrarVolume()
    basic.pause(50)
}

from microbit import *
import music

limiar_palma = 80
limiar_grito = 150
modo = "palma"

def ler_som():
    # Micro:bit V2 tem microfone integrado
    # Em V1, usar pino 0 com sensor de som externo
    try:
        return microphone.sound_level()
    except:
        return pin0.read_analog()  # Fallback para sensor externo

def detectar_tipo_som():
    nivel = ler_som()
    if nivel > limiar_grito:
        return "grito"
    elif nivel > limiar_palma:
        return "palma"
    else:
        return "silêncio"

def reagir_palma():
    display.show(Image.HAPPY)
    # Toca nota musical
    music.play("C4:4")
    sleep(200)
    display.clear()

def reagir_grito():
    display.show(Image.SKULL)
    # Toca alarme
    music.play("C5:8 E5:8 G5:8 C6:8")
    sleep(500)
    display.clear()

def reagir_silencio():
    display.show(Image.HEART)
    sleep(100)
    display.clear()

display.scroll("ORQUESTRA")

while True:
    tipo = detectar_tipo_som()
    
    if modo == "palma":
        if tipo == "palma":
            reagir_palma()
        elif tipo == "grito":
            reagir_grito()
    elif modo == "visual":
        # Mostra volume como barras
        nivel = ler_som()
        barras = int(nivel / 255 * 5)
        display.clear()
        for i in range(barras):
            for j in range(5):
                display.set_pixel(i, j, 9)
        sleep(50)
    else:  # modo silêncio
        if tipo == "silêncio":
            reagir_silencio()
    
    # Mudar modo com botões
    if button_a.was_pressed():
        modo = "palma"
        display.scroll("PALMA")
    if button_b.was_pressed():
        modo = "visual"
        display.scroll("VISUAL")
    if pin0.is_touched():
        modo = "silêncio"
        display.scroll("SILENCIO")
    
    sleep(100)

from microbit import *

def testar_de_morgan():
    # Testa as 4 combinações
    combinacoes = [(False, False), (False, True), (True, False), (True, True)]
    
    for A, B in combinacoes:
        # NOT (A AND B) = (NOT A) OR (NOT B)
        lado1 = not (A and B)
        lado2 = (not A) or (not B)
        
        # Verifica equivalência
        if lado1 == lado2:
            display.show(Image.YES)
        else:
            display.show(Image.NO)
        
        display.scroll(str(A) + " " + str(B))
        sleep(500)
        
        # Mostra na matriz

function desenharGrafico(dados: number[]) {
    display.clear()
    
    // Encontra o máximo para normalizar
    let maximo = 0
    for (let valor of dados) {
        if (valor > maximo) maximo = valor
    }
    if (maximo == 0) maximo = 1
    
    // Desenha eixos
    desenharEixos()
    
    // Desenha barras
    let cores = [9, 7, 5, 3, 1] // Brilho variado
    for (let i = 0; i < dados.length && i < 4; i++) {
        let altura = Math.ceil(dados[i] / maximo * 4)
        let cor = cores[i % cores.length]
        desenharBarra(altura, i + 1, cor)
    }
}

def desenhar_barra(altura, x, brilho=9):
    # Desenha uma barra vertical na matriz

def desenhar_eixos():
    # Desenha eixos X e Y
    for x in range(5):
        display.set_pixel(x, 4, 2)  # Eixo X
    for y in range(5):
        display.set_pixel(0, y, 2)  # Eixo Y

def grafico_barras(dados):
    # Desenha gráfico de barras na matriz 5x5
    # dados: lista de valores (máximo 4 barras)
    display.clear()
    if not dados:
        return
    
    # Normaliza dados (0-5)
    maximo = max(dados)
    if maximo == 0:
        maximo = 1
    
    # Desenha eixos
    desenhar_eixos()
    
    # Desenha barras
    cores = [9, 7, 5, 3]  # Brilho decrescente
    for i in range(min(len(dados), 4)):
        altura = int(dados[i] / maximo * 4) + 1
        desenhar_barra(altura, i + 1, cores[i % len(cores)])
    
    # Mostra valores no display
    for i, valor in enumerate(dados[:4]):
        display.scroll(str(valor))
        sleep(500)

def animar_construcao(dados):
    # Anima a construção do gráfico
    display.clear()
    maximo = max(dados) if max(dados) > 0 else 1
    
    for passo in range(5):
        display.clear()
        desenhar_eixos()
        for i in range(min(len(dados), 4)):
            altura = int(dados[i] / maximo * (passo + 1))
            if altura > 5:
                altura = 5
            desenhar_barra(altura, i + 1, 9)
        sleep(300)

# PROBLEMA: Desperdício de água no bebedouro
# SOLUÇÃO: Robô que apita se a torneira ficar aberta por mais de 5 segundos
# SENSORES: Botão (simula sensor de toque)
# SAÍDAS: Matriz de LEDs (alarme visual)

TEMPO_LIMITE = 5000  # 5 segundos
tempo_inicial = 0
torneira_aberta = False

def alarme():
    # Ativa alarme visual na matriz
    for i in range(3):
        display.show(Image.NO)
        sleep(200)
        display.clear()
        sleep(200)
    display.show(Image.SKULL)

def verificar_torneira():
    global torneira_aberta, tempo_inicial
    # Simula sensor de toque no pino 0
    if pin0.is_touched():
        if not torneira_aberta:
            # Torneira foi aberta
            torneira_aberta = True
            tempo_inicial = running_time()
            display.show(Image.ARROW_E)
    else:
        if torneira_aberta:
            # Torneira foi fechada
            torneira_aberta = False
            display.show(Image.HEART)

def monitorar_tempo():
    global torneira_aberta, tempo_inicial
    if torneira_aberta:
        tempo_corrente = running_time()
        if (tempo_corrente - tempo_inicial) > TEMPO_LIMITE:
            alarme()

# Programa principal
display.scroll("BEBEDOURO")

while True:
    verificar_torneira()
    monitorar_tempo()
